% ============================================================================
% Template API for agents
%
% Content class mapping:
%   main_figure   -> \mainfigure{path}{caption}{label}
%   main_figure   -> \widemainfigure{path}{caption}{label}
%   supp_figure   -> \suppfigure{path}{caption}{label}
%   main_table    -> \maintable{caption}{label}{tabular/body}
%   supp_table    -> \supptable{caption}{label}{tabular/body}
%   appendix_note -> \appendixnote{heading}{body}
%   proof         -> \proofblock{heading}{body}
%   derivation    -> \derivationblock{heading}{body}
%
% Main figures/tables appear where their macro is called. Supplementary
% figures, tables, notes, proofs, and derivations are collected and printed at
% the back by \printsupplement, with separate Figure S1/Table S1 numbering.
% ============================================================================

\NewDocumentCommand{\agentfigureinclude}{m m}{%
  \IfFileExists{#2}{%
    \includegraphics[width=#1]{#2}%
  }{%
    \fbox{%
      \begin{minipage}[c][1.55in][c]{0.72\linewidth}
        \centering\footnotesize Missing figure file:\\\texttt{#2}
      \end{minipage}%
    }%
  }%
}

\NewDocumentCommand{\mainfigure}{m +m m}{%
  \begin{figure}[htbp]
    \centering
    \agentfigureinclude{0.75\textwidth}{#1}
    \captionsetup{width=0.82\textwidth}
    \caption{#2}
    \label{#3}
  \end{figure}
  \FloatBarrier
}

\NewDocumentCommand{\widemainfigure}{m +m m}{%
  \begin{figure}[htbp]
    \centering
    \agentfigureinclude{0.9\textwidth}{#1}
    \captionsetup{width=0.92\textwidth}
    \caption{#2}
    \label{#3}
  \end{figure}
  \FloatBarrier
}

\NewDocumentCommand{\maintable}{m m +m}{%
  \begin{table}[htbp]
    \centering
    \caption{#1}
    \label{#2}
    \small
    #3
  \end{table}
  \FloatBarrier
}

\newcommand{\supplementcontent}{}

\NewDocumentCommand{\suppfigure}{m +m m}{%
  \gappto{\supplementcontent}{%
    \begin{figure}[htbp]
      \centering
      \agentfigureinclude{0.75\textwidth}{#1}
      \captionsetup{width=0.82\textwidth}
      \caption{#2}
      \label{#3}
    \end{figure}
    \FloatBarrier
  }%
}

\NewDocumentCommand{\supptable}{m m +m}{%
  \gappto{\supplementcontent}{%
    \begin{table}[htbp]
      \centering
      \caption{#1}
      \label{#2}
      \small
      #3
    \end{table}
    \FloatBarrier
  }%
}

\NewDocumentCommand{\appendixnote}{m +m}{%
  \gappto{\supplementcontent}{%
    \subsection*{#1}
    #2
  }%
}

\NewDocumentCommand{\proofblock}{m +m}{%
  \gappto{\supplementcontent}{%
    \subsection*{#1}
    \begin{proof}
      #2
    \end{proof}
  }%
}

\NewDocumentCommand{\derivationblock}{m +m}{%
  \gappto{\supplementcontent}{%
    \subsection*{#1}
    #2
  }%
}

\newcommand{\printsupplement}{%
  \ifstrempty{\supplementcontent}{}{%
    \clearpage
    \appendix
    \setcounter{figure}{0}
    \setcounter{table}{0}
    \renewcommand{\thefigure}{S\arabic{figure}}
    \renewcommand{\thetable}{S\arabic{table}}
    \section*{Supplementary Material}
    \addcontentsline{toc}{section}{Supplementary Material}
    \supplementcontent
  }%
}
